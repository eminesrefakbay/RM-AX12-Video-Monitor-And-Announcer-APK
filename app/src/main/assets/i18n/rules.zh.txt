# 每行一条规则:   条件 -> 语音内容
# 可选:  内容 | 条件变为假时播报的内容     ! = 紧急     @10 = 每 10 秒重复
hysteresis = 25
var_hysteresis = 1
hold = 100
initial = off
scale = auto

# --- 开关 (INAV 模式范围, 通道号 = 遥控通道, 1000..2000 us) ---
ch5 >= 1950 -> 已解锁 | 已上锁
ch6 >= 1950 -> 自稳模式
ch6 >= 1450 and ch6 <= 1550 -> 半自稳模式
ch7 >= 1875 -> !返航 | 返航关闭
ch10 <= 1125 or ch10 >= 1875 -> 定点 | 定点关闭
ch10 >= 1950 -> 定高
ch9 <= 1100 -> Airmode 开启 | Airmode 关闭

# --- 电池, 每节电压 (6S): 飞控的数值, 与 OSD 显示一致 ---
# 锂离子电池降落约 3.6 V: 警告 3.70, 严重 3.55。 锂聚合物请用 3.85 / 3.70。
# "vbat > 12" 可在飞控仅由 USB 供电 (vbat = 0) 时避免严重警告。
vbat / 6 < 3.70 and vbat / 6 >= 3.55 -> 电池电量低 @30
vbat / 6 < 3.55 and vbat > 12 -> !电池电量严重不足 @10

# --- 链路: 仅在解锁开关打开时 (ch5), 未通电的飞行器保持安静 ---
up_lq < 70 and ch5 >= 1950 -> 链路质量低
up_lq < 40 and ch5 >= 1950 -> !链路严重不足 @5
tlm == 0 and ch5 >= 1950 -> !遥测丢失

# --- 飞控自身的飞行模式 (请在 设置 > 诊断 > 显示变量 中核对确切名称) ---
fm ~ "RTH" -> 返航已启动
